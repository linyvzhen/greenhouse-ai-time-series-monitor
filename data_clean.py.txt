import pandas as pd

def clean_data(file_path):
    df = pd.read_csv(file_path)
    # 去除空值
    df = df.dropna()
    # 剔除异常值
    df = df[(df["temp"] >= 0) & (df["temp"] <= 50)]
    return df

if __name__ == "__main__":
    data = clean_data("../data/sample_data.csv")
    print("清洗完成，有效数据行数：", len(data))
