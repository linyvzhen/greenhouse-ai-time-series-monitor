import pandas as pd
import numpy as np

def clean_data(df):
    # 缺失值填充
    df["temp"] = df["temp"].fillna(df["temp"].rolling(5).mean())
    df["hum"] = df["hum"].fillna(df["hum"].rolling(5).mean())
    # 异常值过滤
    df = df[(df["temp"]>10)&(df["temp"]<40)]
    # 标准化
    df["temp_norm"] = (df["temp"] - df["temp"].mean()) / df["temp"].std()
    # 构造时段特征、周期特征
    df["hour"] = df["record_time"].dt.hour
    df["dayofweek"] = df["record_time"].dt.dayofweek
    return df

if __name__ == "__main__":
    raw_df = pd.read_csv("../data/greenhouse.csv")
    raw_df["record_time"] = pd.to_datetime(raw_df["record_time"])
    clean_df = clean_data(raw_df)
    clean_df.to_csv("../data/clean_greenhouse.csv",index=False)
    print("数据清洗+特征生成完成")
